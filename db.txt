CREATE TABLE service_categories (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name NVARCHAR(100) NOT NULL,
    slug NVARCHAR(120) NOT NULL UNIQUE,
    description NVARCHAR(500) NULL,
    is_active BIT NOT NULL DEFAULT 1,
    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    updated_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);

CREATE TABLE service_items (
    id INT IDENTITY(1,1) PRIMARY KEY,
    category_id INT NOT NULL,
    name NVARCHAR(150) NOT NULL,
    slug NVARCHAR(180) NOT NULL UNIQUE,
    description NVARCHAR(500) NULL,
    default_price DECIMAL(10,2) NOT NULL DEFAULT 0,
    estimated_minutes INT NULL,
    is_active BIT NOT NULL DEFAULT 1,
    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    updated_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),

    CONSTRAINT FK_service_items_category
        FOREIGN KEY (category_id)
        REFERENCES service_categories(id)
);


CREATE TABLE customers (
    id INT IDENTITY(1,1) PRIMARY KEY,
    name NVARCHAR(150) NOT NULL,
    slug NVARCHAR(180) NOT NULL UNIQUE,

    mobile NVARCHAR(20) NOT NULL,
    alternate_mobile NVARCHAR(20) NULL,
    email NVARCHAR(150) NULL,
    address NVARCHAR(500) NULL,

    is_active BIT NOT NULL DEFAULT 1,
    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    updated_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);

CREATE TABLE vehicles (
    id INT IDENTITY(1,1) PRIMARY KEY,
    customer_id INT NOT NULL,

    slug NVARCHAR(250) NOT NULL UNIQUE,
    registration_number NVARCHAR(30) NOT NULL UNIQUE,

    brand_id INT NULL,
    model_id INT NULL,

    variant NVARCHAR(100) NULL,
    fuel_type NVARCHAR(50) NULL,
    chassis_number NVARCHAR(100) NULL,
    engine_number NVARCHAR(100) NULL,
    manufacture_year INT NULL,

    is_active BIT NOT NULL DEFAULT 1,
    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    updated_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),

    CONSTRAINT FK_vehicles_customer
        FOREIGN KEY (customer_id)
        REFERENCES customers(id)
);

CREATE TABLE gate_entries (
    id INT IDENTITY(1,1) PRIMARY KEY,

    slug NVARCHAR(180) NOT NULL UNIQUE,

    vehicle_id INT NOT NULL,
    customer_id INT NOT NULL,

    entry_number NVARCHAR(50) NOT NULL UNIQUE,

    odometer_reading INT NULL,
    fuel_level NVARCHAR(20) NULL,

    entry_notes NVARCHAR(1000) NULL,

    current_status NVARCHAR(50) NOT NULL DEFAULT 'JOB_CARD_PENDING',

    entry_time DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    exit_time DATETIME2 NULL,

    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    updated_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME()
);

CREATE TABLE gate_entry_photos (
    id INT IDENTITY(1,1) PRIMARY KEY,

    gate_entry_id INT NOT NULL,

    photo_type NVARCHAR(50) NULL,

    photo_url NVARCHAR(500) NOT NULL,

    uploaded_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),

    CONSTRAINT FK_gate_entry_photos
        FOREIGN KEY (gate_entry_id)
        REFERENCES gate_entries(id)
);

CREATE TABLE job_cards (
    id INT IDENTITY(1,1) PRIMARY KEY,

    job_card_no NVARCHAR(30) NOT NULL UNIQUE,

    gate_entry_id INT NOT NULL,
    customer_id INT NOT NULL,
    vehicle_id INT NOT NULL,

    expected_delivery_at DATETIME2 NULL,

    additional_notes NVARCHAR(MAX) NULL,

    total_estimate DECIMAL(10,2) NOT NULL DEFAULT 0,

    status NVARCHAR(30) NOT NULL DEFAULT 'PENDING_APPROVAL',

    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),
    updated_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),

    CONSTRAINT FK_job_cards_gate_entry
        FOREIGN KEY (gate_entry_id) REFERENCES gate_entries(id),

    CONSTRAINT FK_job_cards_customer
        FOREIGN KEY (customer_id) REFERENCES customers(id),

    CONSTRAINT FK_job_cards_vehicle
        FOREIGN KEY (vehicle_id) REFERENCES vehicles(id)
);


CREATE TABLE job_card_services (
    id INT IDENTITY(1,1) PRIMARY KEY,

    job_card_id INT NOT NULL,
    service_id INT NOT NULL,

    service_name NVARCHAR(150) NOT NULL,
    price DECIMAL(10,2) NOT NULL,

    status NVARCHAR(30) NOT NULL DEFAULT 'PENDING',

    created_at DATETIME2 NOT NULL DEFAULT SYSUTCDATETIME(),

    CONSTRAINT FK_job_card_services_job_card
        FOREIGN KEY (job_card_id) REFERENCES job_cards(id),

    CONSTRAINT FK_job_card_services_service
        FOREIGN KEY (service_id) REFERENCES service_master(id)
);
